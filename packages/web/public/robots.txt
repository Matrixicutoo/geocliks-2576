# GeoCliks — https://geocliks.com

# The marketing site, the Help Center and the legal pages are meant to be
# indexed. Everything else here is either behind a sign-in or reachable only
# with a token, and none of it belongs in search results.
#
# Disallow is a crawl instruction, not an access control: a disallowed URL that
# someone links to publicly can still be indexed without being fetched. The
# pages below therefore also send `<meta name="robots" content="noindex">` at
# runtime (see src/web/lib/seo.ts, useNoindex).

User-agent: *

# Signed-in application and workspace admin.
Disallow: /app
Disallow: /admin

# Token and code URLs. These address one customer's evidence — a share link, a
# live delivery track, a single photo's verification page, a workspace invite.
# Indexing them would publish the token itself in search results.
Disallow: /share/
Disallow: /t/
Disallow: /v/
Disallow: /join/
Disallow: /verify

# Account recovery.
Disallow: /reset-password

# Auth screens. No content to rank, and they duplicate each other's copy.
Disallow: /sign-in
Disallow: /sign-up

# /pricing is its own page now — the plans, their limits and the comparison table
# all live there, and "/#pricing" forwards to it. Crawlable on purpose: it is a URL
# people type, link to and search for.

# Answer engines and AI crawlers, named explicitly.
#
# They are already covered by the `User-agent: *` group above; these groups exist
# to say so on purpose rather than by default, because /blog is written to be
# quoted by them. A named group replaces the wildcard for that crawler entirely,
# so each one repeats the private-URL rules — otherwise naming GPTBot here would
# hand it /app and every share token.
#
# Google-Extended is not a crawler and takes no Disallow of its own: it is the
# switch that says Gemini and the AI features in Search may use what Googlebot
# already fetched. Allow is the answer we want.

User-agent: GPTBot
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: PerplexityBot
User-agent: Perplexity-User
User-agent: ClaudeBot
User-agent: Claude-User
User-agent: Applebot-Extended
User-agent: CCBot
Allow: /
Disallow: /app
Disallow: /admin
Disallow: /share/
Disallow: /t/
Disallow: /v/
Disallow: /join/
Disallow: /verify
Disallow: /reset-password
Disallow: /sign-in
Disallow: /sign-up

User-agent: Google-Extended
Allow: /
